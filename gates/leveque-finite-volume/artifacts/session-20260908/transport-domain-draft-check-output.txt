NumStability.leveque01_equation03_solutionDomains (profile : ℝ → ℝ) (speed : ℝ) :
  (∀ (x : ℝ), NumStability.travelingWave profile speed x 0 = profile x) ∧
    (∀ (x t : ℝ), NumStability.travelingWave profile speed (x + speed * t) t = profile x) ∧
      (∀ (x t : ℝ), HasDerivAt (fun τ => NumStability.travelingWave profile speed (x + speed * τ) τ) 0 t) ∧
        (NumStability.IsLinearAdvectionSolution (NumStability.travelingWave profile speed) speed ↔
            Differentiable ℝ profile) ∧
          ((NumStability.IsRectangleConservationLawSolution (NumStability.travelingWave profile speed) fun state =>
              speed * state) ↔
            ∀ (a b : ℝ), IntervalIntegrable profile volume a b)
