usage: reconciliation_launcher.py validate [-h] --request REQUEST
                                           [--status STATUS] --epoch EPOCH
                                           [--integration-receipt INTEGRATION_RECEIPT]
                                           [--stable-promotion-receipt STABLE_PROMOTION_RECEIPT]
                                           [--validation-command-timeout SECONDS]
                                           [--timeout SECONDS]

options:
  -h, --help            show this help message and exit
  --request REQUEST
  --status STATUS
  --epoch EPOCH
  --integration-receipt INTEGRATION_RECEIPT
  --stable-promotion-receipt STABLE_PROMOTION_RECEIPT
  --validation-command-timeout SECONDS
                        maximum runtime for each validation inside pristine
                        replay (default: 21600)
  --timeout SECONDS     maximum runtime for the complete reconciliation
                        verifier (default: 86400)
