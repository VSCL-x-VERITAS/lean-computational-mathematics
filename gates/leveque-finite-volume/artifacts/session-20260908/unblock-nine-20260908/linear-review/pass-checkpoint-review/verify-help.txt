usage: reconciliation.py verify [-h] --topology TOPOLOGY --epoch EPOCH
                                [--resolve-repositories]
                                [--for-stable-promotion]
                                [--integration-receipt INTEGRATION_RECEIPT]
                                [--stable-promotion-receipt STABLE_PROMOTION_RECEIPT]
                                [--candidate-repository CANDIDATE_REPOSITORY]
                                [--allow-candidate-destination]
                                [--validation-command-timeout SECONDS]

options:
  -h, --help            show this help message and exit
  --topology TOPOLOGY
  --epoch EPOCH
  --resolve-repositories
  --for-stable-promotion
  --integration-receipt INTEGRATION_RECEIPT
  --stable-promotion-receipt STABLE_PROMOTION_RECEIPT
  --candidate-repository CANDIDATE_REPOSITORY
                        read-only scratch repository containing the exact pre-
                        admission candidate
  --allow-candidate-destination
                        allow only the candidate instance ref to have advanced
                        to the exact epoch candidate
  --validation-command-timeout SECONDS
                        maximum runtime for each replayed validation command
                        (default: 21600)
