            [inst_1 : TopologicalSpace Point] →

                    D → (Point → ℝ → Fin m → ℝ) → ℝ → ℝ → Prop :=

      ∀ τ ∈ Set.uIcc s t, MeasureTheory.IntegrableOn (fun x => q x τ) (data.cells.cellRegion cell) data.measure) ∧
    (∀ (cell : Cell),

          MeasureTheory.Integrable
              (fun point =>

              (data.faceMeasure d (data.rightFace d cell))) ∧
      (∀ x ∈ data.cells.domain, ∀ τ ∈ Set.uIcc s t, q x τ ∈ data.admissibleStates d) ∧

                    v) ∧
