ScopedRiemannInformationDraft.Method.reference_comparison.{u_1, u_2} {m : ℕ} {law : ScopedRiemannInformationDraft.Law m}
  {Result : ScopedRiemannInformationDraft.Problem law → Type u_1} {Information : Type u_2}
  (method : ScopedRiemannInformationDraft.Method law Result Information)
  (problem : ScopedRiemannInformationDraft.Problem law) (hdomain : method.domain problem) :
  ∃ reference,
    NumStability.IsRiemannData (fun x => reference.field x 0) problem.left problem.right ∧
      NumStability.IsOneDimensionalCellAverage (fun τ => law.flux (reference.field 0 τ)) 0 problem.duration
          reference.meanFlux ∧
        0 ≤ method.errorBound problem ∧
          ‖method.flux problem hdomain - reference.meanFlux‖ ≤ method.errorBound problem ∧
            ∀ (physicalMean : Fin m → ℝ) (comparisonBound : ℝ),
              ‖reference.meanFlux - physicalMean‖ ≤ comparisonBound →
                ‖method.flux problem hdomain - physicalMean‖ ≤ method.errorBound problem + comparisonBound
'ScopedRiemannInformationDraft.Method.reference_comparison' depends on axioms: [propext, Classical.choice, Quot.sound]
