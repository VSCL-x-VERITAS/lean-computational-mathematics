NumStability.ProspectiveSourceAlternatives.density_rectangle_capstone {m : ℕ} {q : ℝ → ℝ → Fin (m + 1) → ℝ}
  {flux : (Fin (m + 1) → ℝ) → Fin (m + 1) → ℝ} {production : ℝ → ℝ → Fin (m + 1) → ℝ}
  (h : IsRectangleBalanceLawSolution q flux production) :
  IsRectangleBalanceLawSolution q flux production ∧
    (∀ (a b s t : ℝ),
        ∫ (τ : ℝ) in s..t, ∫ (x : ℝ) in a..b, production x τ =
          ((∫ (x : ℝ) in a..b, q x t) - ∫ (x : ℝ) in a..b, q x s) - ∫ (τ : ℝ) in s..t, flux (q a τ) - flux (q b τ)) ∧
      (IsRectangleConservationLawSolution q flux ↔
          ∀ (a b s t : ℝ), ∫ (τ : ℝ) in s..t, ∫ (x : ℝ) in a..b, production x τ = 0) ∧
        (¬IsRectangleConservationLawSolution q flux ↔
            ∃ a b s t, ∫ (τ : ℝ) in s..t, ∫ (x : ℝ) in a..b, production x τ ≠ 0) ∧
          ∀ (a b : ℝ),
            ∀ᵐ (t : ℝ),
              HasDerivAt (fun s => ∫ (x : ℝ) in a..b, q x s)
                (flux (q a t) - flux (q b t) + ∫ (x : ℝ) in a..b, production x t) t
NumStability.ProspectiveSourceAlternatives.signed_density_fixture (rate : ℝ) :
  (IsRectangleBalanceLawSolution (fun x t => rate * t * riemannData 0 0 1 x) (fun x => 0) fun x x_1 =>
      rate * riemannData 0 0 1 x) ∧
    ∫ (_τ : ℝ) (x : ℝ) in 1..2, rate * riemannData 0 0 1 x = rate ∧
      (rate ≠ 0 → ¬IsRectangleConservationLawSolution (fun x t => rate * t * riemannData 0 0 1 x) fun x => 0)
NumStability.ProspectiveSourceAlternatives.nonsmooth_density_fixture :
  (IsRectangleBalanceLawSolution (fun x t => t * riemannData 0 0 1 x) (fun x => 0) fun x x_1 => riemannData 0 0 1 x) ∧
    ¬ContinuousAt (fun x => 1 * riemannData 0 0 1 x) 0 ∧
      (∀ (a b t : ℝ),
          HasDerivAt (fun τ => ∫ (x : ℝ) in a..b, τ * riemannData 0 0 1 x) (∫ (x : ℝ) in a..b, riemannData 0 0 1 x) t) ∧
        ∫ (_τ : ℝ) (x : ℝ) in 1..2, riemannData 0 0 1 x = 1 ∧
          ¬IsRectangleConservationLawSolution (fun x t => t * riemannData 0 0 1 x) fun x => 0
def NumStability.ProspectiveSourceAlternatives.LocalMediumAlternative.{u_1, u_2} : {M : Type u_1} →
  {Q : Type u_2} → [TopologicalSpace M] → (ℝ → M) → (ℝ → Q) → M → M → Q → Q → Prop :=
fun {M} {Q} [TopologicalSpace M] medium initialState ml mr ql qr =>
  HasJumpAt medium 0 ml mr ∧ IsRiemannData initialState ql qr ∧ ql ≠ qr
def NumStability.ProspectiveSourceAlternatives.GlobalMediumAlternative.{u_1, u_2} : {M : Type u_1} →
  {Q : Type u_2} → (ℝ → M) → (ℝ → Q) → M → M → Q → Q → Prop :=
fun {M} {Q} medium initialState ml mr ql qr =>
  IsRiemannData medium ml mr ∧ ml ≠ mr ∧ IsRiemannData initialState ql qr ∧ ql ≠ qr
NumStability.ProspectiveSourceAlternatives.local_medium_capstone.{u_1, u_2} {M : Type u_1} {Q : Type u_2}
  [TopologicalSpace M] [TopologicalSpace Q] [T2Space M] [T2Space Q] {medium : ℝ → M} {initialState : ℝ → Q} {ml mr : M}
  {ql qr : Q} (h : LocalMediumAlternative medium initialState ml mr ql qr) :
  (HasJumpAt medium 0 ml mr ∧ HasJumpAt initialState 0 ql qr) ∧
    IsRiemannData initialState ql qr ∧
      ¬ContinuousAt medium 0 ∧
        ¬ContinuousAt initialState 0 ∧ ∃ originState, initialState = riemannData ql originState qr
NumStability.ProspectiveSourceAlternatives.global_medium_capstone.{u_1, u_2} {M : Type u_1} {Q : Type u_2}
  [TopologicalSpace M] [TopologicalSpace Q] {medium : ℝ → M} {initialState : ℝ → Q} {ml mr : M} {ql qr : Q}
  (h : GlobalMediumAlternative medium initialState ml mr ql qr) :
  LocalMediumAlternative medium initialState ml mr ql qr ∧
    IsRiemannData (fun x => (medium x, initialState x)) (ml, ql) (mr, qr) ∧
      ml ≠ mr ∧
        ql ≠ qr ∧
          (∃ originMaterial, medium = riemannData ml originMaterial mr) ∧
            ∃ originState, initialState = riemannData ql originState qr
NumStability.ProspectiveSourceAlternatives.global_medium_fixture (originMaterial originState : ℝ) :
  GlobalMediumAlternative (riemannData 0 originMaterial 1) (riemannData 2 originState 3) 0 1 2 3 ∧
    riemannData 0 originMaterial 1 0 = originMaterial ∧ riemannData 2 originState 3 0 = originState
NumStability.ProspectiveSourceAlternatives.local_medium_separating_fixture (originMaterial originState : ℝ) :
  LocalMediumAlternative (LocalMaterialInterface.varyingMedium originMaterial) (riemannData 2 originState 3) 0 1 2 3 ∧
    (¬∃ ml mr, IsRiemannData (LocalMaterialInterface.varyingMedium originMaterial) ml mr) ∧
      LocalMaterialInterface.varyingMedium originMaterial 0 = originMaterial ∧
        riemannData 2 originState 3 0 = originState
NumStability.ProspectiveSourceAlternatives.positive_dimensional_problem_capstone {m : ℕ}
  (problem : FirstOrderInitialValueProblem (Fin (m + 1))) :
  problem.IsRiemann ↔
    (∃ leftState rightState,
        problem.governing.IsHyperbolic ∧
          leftState ∈ problem.governing.admissibleStates ∧
            rightState ∈ problem.governing.admissibleStates ∧
              (∀ x < 0, problem.initialState x = leftState) ∧ ∀ (x : ℝ), 0 < x → problem.initialState x = rightState) ∧
      ∀ (x t : ℝ) (state qt qx : Fin (m + 1) → ℝ),
        problem.governing.residual x t state qt qx = 0 ↔
          qt + (problem.governing.principal x t state).mulVec qx = problem.governing.forcing x t state
NumStability.ProspectiveSourceAlternatives.equal_or_distinct_initial_states {m : ℕ}
  (equation : FirstOrderEquation (Fin (m + 1))) (hhyper : equation.IsHyperbolic)
  (leftState origin rightState : Fin (m + 1) → ℝ) (hleft : leftState ∈ equation.admissibleStates)
  (hright : rightState ∈ equation.admissibleStates) :
  (fromStates equation leftState origin rightState).IsRiemannWithStates leftState rightState ∧
    (leftState = rightState → ¬(fromStates equation leftState origin rightState).IsJumpRiemann) ∧
      (leftState ≠ rightState → (fromStates equation leftState origin rightState).IsJumpRiemann)
def NumStability.ProspectiveSourceAlternatives.scalarEquation : FirstOrderEquation (Fin 1) :=
FirstOrderEquation.constantLinear 1
NumStability.ProspectiveSourceAlternatives.scalarEquation_hyperbolic : scalarEquation.IsHyperbolic
NumStability.ProspectiveSourceAlternatives.scalar_problem_fixture (origin : Fin 1 → ℝ) :
  (fromStates scalarEquation (fun x => 0) origin fun x => 1).IsJumpRiemann ∧
    (fromStates scalarEquation (fun x => 0) origin fun x => 0).IsRiemann ∧
      ¬(fromStates scalarEquation (fun x => 0) origin fun x => 0).IsJumpRiemann ∧
        ∀ (x t : ℝ) (state qt qx : Fin 1 → ℝ), scalarEquation.residual x t state qt qx = 0 ↔ qt + qx = 0
def NumStability.IsRectangleBalanceLawSolution.{u_1} : {E : Type u_1} →
  [inst : NormedAddCommGroup E] → [NormedSpace ℝ E] → (ℝ → ℝ → E) → (E → E) → (ℝ → ℝ → E) → Prop :=
fun {E} [NormedAddCommGroup E] [NormedSpace ℝ E] q flux production =>
  (∀ (a b t : ℝ), IntervalIntegrable (fun x => q x t) volume a b) ∧
    (∀ (x s t : ℝ), IntervalIntegrable (fun τ => flux (q x τ)) volume s t) ∧
      (∀ (a b t : ℝ), IntervalIntegrable (fun x => production x t) volume a b) ∧
        (∀ (a b s t : ℝ), IntervalIntegrable (fun τ => ∫ (x : ℝ) in a..b, production x τ) volume s t) ∧
          ∀ (a b s t : ℝ),
            (∫ (x : ℝ) in a..b, q x t) - ∫ (x : ℝ) in a..b, q x s =
              (∫ (τ : ℝ) in s..t, flux (q a τ) - flux (q b τ)) + ∫ (τ : ℝ) in s..t, ∫ (x : ℝ) in a..b, production x τ
def NumStability.IsRectangleConservationLawSolution.{u_1} : {E : Type u_1} →
  [inst : NormedAddCommGroup E] → [NormedSpace ℝ E] → (ℝ → ℝ → E) → (E → E) → Prop :=
fun {E} [NormedAddCommGroup E] [NormedSpace ℝ E] q flux =>
  (∀ (a b t : ℝ), IntervalIntegrable (fun x => q x t) volume a b) ∧
    (∀ (x s t : ℝ), IntervalIntegrable (fun τ => flux (q x τ)) volume s t) ∧
      ∀ (a b s t : ℝ),
        (∫ (x : ℝ) in a..b, q x t) - ∫ (x : ℝ) in a..b, q x s = ∫ (τ : ℝ) in s..t, flux (q a τ) - flux (q b τ)
def NumStability.HasJumpAt.{u_1} : {E : Type u_1} → [TopologicalSpace E] → (ℝ → E) → ℝ → E → E → Prop :=
fun {E} [TopologicalSpace E] data a left right =>
  Tendsto data (𝓝[<] a) (𝓝 left) ∧ Tendsto data (𝓝[>] a) (𝓝 right) ∧ left ≠ right
def NumStability.IsRiemannData.{u_1} : {State : Type u_1} → (ℝ → State) → State → State → Prop :=
fun {State} data leftState rightState => (∀ x < 0, data x = leftState) ∧ ∀ (x : ℝ), 0 < x → data x = rightState
def NumStability.riemannData.{u_1} : {State : Type u_1} → State → State → State → ℝ → State :=
fun {State} leftState valueAtOrigin rightState x =>
  if x < 0 then leftState else if 0 < x then rightState else valueAtOrigin
