          adapted.interfaceFlux old hdomain j = NumStability.rectangleRiemannInterfaceFlux method old hdomain j

NumStability.ReturnedFieldInterfaceDraft.nonexact_method_with_reference :
  ∃ problem,
    NumStability.StationaryRiemannField.method.domain problem ∧
      NumStability.IsRiemannData
          (fun x =>
            NumStability.StationaryRiemannField.method.field
              (NumStability.StationaryRiemannField.method.solve problem trivial) x 0)
          problem.leftState problem.rightState ∧
        ¬NumStability.IsRectangleConservationLawSolution
              (NumStability.StationaryRiemannField.method.field
                (NumStability.StationaryRiemannField.method.solve problem trivial))
              NumStability.StationaryRiemannField.transportLaw.physicalFlux ∧
          NumStability.IsRiemannData
              (fun x => NumStability.StationaryRiemannField.reference problem.leftState problem.rightState x 0)
              problem.leftState problem.rightState ∧
            NumStability.IsRectangleConservationLawSolution
                (NumStability.StationaryRiemannField.reference problem.leftState problem.rightState)
                NumStability.StationaryRiemannField.transportLaw.physicalFlux ∧
              ∀ (t : ℝ),
                0 < t →
                  NumStability.StationaryRiemannField.method.numericalFlux
                      (NumStability.StationaryRiemannField.method.extract
                        (NumStability.StationaryRiemannField.method.solve problem trivial)) =
                    NumStability.StationaryRiemannField.transportLaw.physicalFlux
                      (NumStability.StationaryRiemannField.reference problem.leftState problem.rightState 0 t)
